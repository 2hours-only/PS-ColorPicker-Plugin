console.log("✅ LUT Picker V5 loaded"); 

const { core, app } = require("photoshop");

const svArea = document.getElementById("svArea");
const hueArea = document.getElementById("hueArea");
const svCanvas = document.getElementById("svCanvas");
const hueCanvas = document.getElementById("hueCanvas");
const svCtx = svCanvas.getContext("2d");
const hueCtx = hueCanvas.getContext("2d");
const svMarker = document.getElementById("svMarker");

let currentHue = 0;
let currentS = 1;
let currentV = 1;
let cycleState = 0; 
let targetGray = null;

const LUT = {
  gray: null,
  sv: null,
  ready: false,
  H: 45,
  S: 128,
  G: 128,
  HF: 4,
  SF: 2,
  async init() {
    try {
      const [grayBuf, svBuf] = await Promise.all([
        fetch("./asset/hsv_to_gray_down.bin").then((r) => r.arrayBuffer()),
        fetch("./asset/hsg_to_sv_down.bin").then((r) => r.arrayBuffer()),
      ]);
      this.gray = new Uint8Array(grayBuf);
      this.sv = new Uint8Array(svBuf);
      this.ready = true;
      console.log(
        "✅ LUT loaded gray:",
        this.gray.length,
        " sv:",
        this.sv.length,
      );
      redraw(true);
    } catch (e) {
      console.warn("❌ LUT load failed:", e);
    }
  },
  _lerp3(arr, d0, d1, d2, f0, f1, f2) {
    const i0 = f0 | 0,
      i1 = f1 | 0,
      i2 = f2 | 0;
    const j0 = Math.min(i0 + 1, d0 - 1);
    const j1 = Math.min(i1 + 1, d1 - 1);
    const j2 = Math.min(i2 + 1, d2 - 1);
    const a = f0 - i0,
      b = f1 - i1,
      c = f2 - i2;
    const x = 1 - a,
      y = 1 - b,
      z = 1 - c;
    const p = (a0, a1, a2) => a0 * d1 * d2 + a1 * d2 + a2;
    return (
      arr[p(i0, i1, i2)] * x * y * z +
      arr[p(j0, i1, i2)] * a * y * z +
      arr[p(i0, j1, i2)] * x * b * z +
      arr[p(j0, j1, i2)] * a * b * z +
      arr[p(i0, i1, j2)] * x * y * c +
      arr[p(j0, i1, j2)] * a * y * c +
      arr[p(i0, j1, j2)] * x * b * c +
      arr[p(j0, j1, j2)] * a * b * c
    );
  },
  _lerp3c2(arr, d0, d1, d2, f0, f1, f2) {
    const i0 = f0 | 0,
      i1 = f1 | 0,
      i2 = f2 | 0;
    const j0 = Math.min(i0 + 1, d0 - 1);
    const j1 = Math.min(i1 + 1, d1 - 1);
    const j2 = Math.min(i2 + 1, d2 - 1);
    const a = f0 - i0,
      b = f1 - i1,
      c = f2 - i2;
    const x = 1 - a,
      y = 1 - b,
      z = 1 - c;
    const p = (a0, a1, a2, ch) => (a0 * d1 * d2 + a1 * d2 + a2) * 2 + ch;
    const res = [0, 0];
    for (let ch = 0; ch < 2; ch++) {
      res[ch] = Math.round(
        arr[p(i0, i1, i2, ch)] * x * y * z +
          arr[p(j0, i1, i2, ch)] * a * y * z +
          arr[p(i0, j1, i2, ch)] * x * b * z +
          arr[p(j0, j1, i2, ch)] * a * b * z +
          arr[p(i0, i1, j2, ch)] * x * y * c +
          arr[p(j0, i1, j2, ch)] * a * y * c +
          arr[p(i0, j1, j2, ch)] * x * b * c +
          arr[p(j0, j1, j2, ch)] * a * b * c,
      );
    }
    return res;
  },
  _hf(hueDeg) {
    return Math.max(0, Math.min(this.H - 1, hueDeg / (2 * this.HF)));
  },
  _sf(v01) {
    return Math.max(0, Math.min(this.S - 1, (v01 * 255) / this.SF));
  },
  _gf(gray) {
    return Math.max(0, Math.min(this.G - 1, gray / this.SF));
  },
  grayFromHsv(hueDeg, s01, v01) {
    if (!this.ready) return null;
    return this._lerp3(
      this.gray,
      this.H,
      this.S,
      this.S,
      this._hf(hueDeg),
      this._sf(s01),
      this._sf(v01),
    );
  },
  svFromHsg(hueDeg, s01, gray) {
    if (!this.ready) return null;
    return this._lerp3c2(
      this.sv,
      this.H,
      this.S,
      this.G,
      this._hf(hueDeg),
      this._sf(s01),
      this._gf(gray),
    );
  },
};

if (debugBtn) {
  debugBtn.addEventListener("click", () => {
    cycleState = (cycleState + 1) % 2;
    if (cycleState === 0) {
      console.log("🟡 状态 0：PS标准取色器");

      console.log("🔴 状态 1：等明度锁定");
      targetGray = perceptualBrightness(currentHue, currentS, currentV);
    }
    redraw(true);
  });
}

function hsvToRgb(h, s, v) {
  const c = v * s;
  const x = c * (1 - Math.abs(((h / 60) % 2) - 1));
  const m = v - c;
  let r = 0,
    g = 0,
    b = 0;
  if (h < 60) {
    r = c;
    g = x;
  } else if (h < 120) {
    r = x;
    g = c;
  } else if (h < 180) {
    g = c;
    b = x;
  } else if (h < 240) {
    g = x;
    b = c;
  } else if (h < 300) {
    r = x;
    b = c;
  } else {
    r = c;
    b = x;
  }
  return {
    r: Math.round((r + m) * 255),
    g: Math.round((g + m) * 255),
    b: Math.round((b + m) * 255),
  };
}

function rgbToHex(r, g, b) {
  return "#" + [r, g, b].map((c) => c.toString(16).padStart(2, "0")).join("");
}

function perceptualBrightness(h, s, v) {
  const lutVal = LUT.grayFromHsv(h, s, v);
  if (lutVal !== null) return Math.round(lutVal);
  const { r, g, b } = hsvToRgb(h, s, v);
  return Math.round(0.299 * r + 0.587 * g + 0.114 * b);
}

function drawSVBackground() {
  const w = svCanvas.width;
  const h = svCanvas.height;
  if (w <= 0 || h <= 0) return;
  svCtx.clearRect(0, 0, w, h);
  const pureRgb = hsvToRgb(currentHue, 1, 1);
  svCtx.fillStyle = rgbToHex(pureRgb.r, pureRgb.g, pureRgb.b);
  svCtx.fillRect(0, 0, w, h);

  const whiteGrad = svCtx.createLinearGradient(0, 0, w, 0);
  whiteGrad.addColorStop(0, "rgba(255,255,255,1)");
  whiteGrad.addColorStop(1, "rgba(255,255,255,0)");
  svCtx.fillStyle = whiteGrad;
  svCtx.fillRect(0, 0, w, h);

  const blackGrad = svCtx.createLinearGradient(0, 0, 0, h);
  blackGrad.addColorStop(0, "rgba(0,0,0,0)");
  blackGrad.addColorStop(1, "rgba(0,0,0,1)");
  svCtx.fillStyle = blackGrad;
  svCtx.fillRect(0, 0, w, h);
}

function drawHueBackground() {
  const w = hueCanvas.width;
  const h = hueCanvas.height;
  if (w <= 0 || h <= 0) return;
  const grad = hueCtx.createLinearGradient(0, h, 0, 0);
  grad.addColorStop(0, "#ff0000");
  grad.addColorStop(1 / 6, "#ffff00");
  grad.addColorStop(2 / 6, "#00ff00");
  grad.addColorStop(3 / 6, "#00ffff");
  grad.addColorStop(4 / 6, "#0000ff");
  grad.addColorStop(5 / 6, "#ff00ff");
  grad.addColorStop(1, "#ff0000");
  hueCtx.fillStyle = grad;
  hueCtx.fillRect(0, 0, w, h);
}

function updateMarkers() {
  const svW = svArea.clientWidth;
  const svH = svArea.clientHeight;
  const hueH = hueArea.clientHeight;
  svMarker.style.transform = `translate(${currentS * svW}px, ${(1 - currentV) * svH}px) translate(-50%, -50%)`;
  hueMarker.style.transform = `translateY(${(1 - currentHue / 360) * hueH}px) translateY(-50%)`;
}

function redraw(needsHueUpdate = true) {
  if (needsHueUpdate) {
    drawSVBackground();
  }
  updateMarkers();
  updateColorInfo();
}

function updateColorInfo() {
  const rgb = hsvToRgb(currentHue, currentS, currentV);
  const hex = rgbToHex(rgb.r, rgb.g, rgb.b);
  swatch.style.backgroundColor = hex;
  colorText.textContent =
    "H:" +
    Math.round(currentHue) +
    " S:" +
    Math.round(currentS * 100) +
    "%" +
    " V:" +
    Math.round(currentV * 100) +
    "%" +
    " " +
    hex;
}

function resizeCanvases() {
  const svRect = svArea.getBoundingClientRect();
  const hueRect = hueArea.getBoundingClientRect();
  if (svRect.width > 0 && svRect.height > 0) {
    svCanvas.width = Math.round(svRect.width);
    svCanvas.height = Math.round(svRect.height);
  }
  if (hueRect.width > 0 && hueRect.height > 0) {
    hueCanvas.width = Math.round(hueRect.width);
    hueCanvas.height = Math.round(hueRect.height);
  }
  drawHueBackground();
  redraw(true);
}

let draggingSV = false;
let draggingHue = false;

svArea.addEventListener("mousedown", (e) => {
  draggingSV = true;
  updateSVFromMouse(e);
});
svArea.addEventListener("mousemove", (e) => {
  if (!draggingSV) return;
  updateSVFromMouse(e);
});
hueArea.addEventListener("mousedown", (e) => {
  draggingHue = true;
  updateHueFromMouse(e);
});
hueArea.addEventListener("mousemove", (e) => {
  if (!draggingHue) return;
  updateHueFromMouse(e);
});

document.body.addEventListener("mouseup", async () => {
  if (draggingSV || draggingHue) {
    const rgb = hsvToRgb(currentHue, currentS, currentV);
    console.log("[Debug] mouseup triggered. Target RGB:", rgb);

    try {
      await core.executeAsModal(
        async () => {

          const newColor = app.foregroundColor;
          newColor.rgb.red = rgb.r;
          newColor.rgb.green = rgb.g;
          newColor.rgb.blue = rgb.b;
          app.foregroundColor = newColor;
          console.log("[Debug] executeAsModal success!");
        },
        { commandName: "Set Foreground Color" },
      );
    } catch (e) {

      console.error(
        "[Debug] Set PS foreground color failed. Full error object:",
        e,
      );
    }
  }
  draggingSV = false;
  draggingHue = false;
});

function updateSVFromMouse(e) {
  const rect = svArea.getBoundingClientRect();
  const s = Math.max(0, Math.min(1, (e.clientX - rect.left) / rect.width));

  if (cycleState === 1 && LUT.ready) {
    const [sPrime, vVal] = LUT.svFromHsg(currentHue, s, targetGray);
    currentS = s;
    currentV = Math.max(0, Math.min(1, vVal / 255));
  } else {
    currentS = s;
    currentV = Math.max(
      0,
      Math.min(1, 1 - (e.clientY - rect.top) / rect.height),
    );
  }
  redraw(false);
}

function updateHueFromMouse(e) {
  const rect = hueArea.getBoundingClientRect();
  const h = Math.max(
    0,
    Math.min(360, (1 - (e.clientY - rect.top) / rect.height) * 360),
  );

  if (cycleState === 1 && LUT.ready) {
    const [sPrime, vVal] = LUT.svFromHsg(h, currentS, targetGray);
    currentHue = h;
    currentV = Math.max(0, Math.min(1, vVal / 255));
  } else {
    currentHue = h;
  }
  redraw(true);
}

resizeCanvases();
if (typeof ResizeObserver !== "undefined") {
  const ro = new ResizeObserver(() => {
    resizeCanvases();
  });
  ro.observe(svArea);
  ro.observe(hueArea);
}
LUT.init();
